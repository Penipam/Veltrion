`;
    const b64 = b64Raw.replace(/\s+/g, '');
    const img = new Image();
    img.src = "data:image/png;base64," + b64;
    img.onload = () => {
      const canvas = document.getElementById("canvas");
      canvas.width = img.width;
      canvas.height = img.height;
      const ctx = canvas.getContext("2d");
      ctx.drawImage(img, 0, 0);

      const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
      const data = imgData.data;

      // Make all background pixels 100% transparent
      for (let i = 0; i < data.length; i += 4) {
        const r = data[i];
        const g = data[i + 1];
        const b = data[i + 2];
        const brightness = (r + g + b) / 3;

        if (brightness < 180) {
          data[i + 3] = 0; // Fully transparent
        } else {
          // Pure white
          data[i] = 255;
          data[i + 1] = 255;
          data[i + 2] = 255;
          data[i + 3] = 255;
        }
      }

      ctx.putImageData(imgData, 0, 0);

      // Export clean transparent PNG
      const dataUrl = canvas.toDataURL("image/png");
      
      const a = document.createElement("a");
      a.id = "downloadLink";
      a.href = dataUrl;
      a.download = "vp_logo_detoure_final.png";
      a.innerHTML = "Download Clean Logo";
      document.body.appendChild(a);

      const ta = document.createElement("textarea");
      ta.id = "dataUrlText";
      ta.style.width = "100%";
      ta.style.height = "200px";
      ta.value = dataUrl;
      document.body.appendChild(ta);

      console.log("PROCESSED_LOGO_READY");
      document.body.innerHTML += "<h3 id='ready'>DONE</h3>";
    };
  </script>
</body>
